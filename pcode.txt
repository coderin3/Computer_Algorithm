A ← [24, 75, 92, 83, 61, 48, 97, 50]


// Type 1
// Input:  array A, size n
// Output: (MIN, MAX)

procedure MaxMin_Type1(A, n)

    min ← A[0]
    max ← A[0]

    for i ← 1 to n-1 do
        if A[i] < min then min ← A[i]
        if A[i] > max then max ← A[i]

    return (min, max)                          2(n-1) comparisons


// Type 2
// Input:  array A, left index i, right index j
// Output: (MIN, MAX)

procedure MaxMin(A, i, j)

    1. if i = j then return (A[i], A[i])       No comparisons needed

    2. if j-i = 1 then if A[i] < A[j] then return (A[i], A[j])
                       else return (A[j], A[i])    One comparison needed

    3. else
       (minLeft, maxLeft)   ← MaxMin(A, i, ⌊(i+j)/2⌋)
       (minRight, maxRight) ← MaxMin(A, ⌊(i+j)/2⌋ + 1, j)

       return (min(minLeft, minRight), max(maxLeft, maxRight))
                                               Two comparisons needed


// Run
A ← [24, 75, 92, 83, 61, 48, 97, 50],  n ← 8

(min, max) ← MaxMin_Type1(A, n)
print min, max        expected comparisons: 2n-2 = 14

(min, max) ← MaxMin(A, 0, n-1)
print min, max        expected comparisons: ⌊3n/2⌋-2 = 10
